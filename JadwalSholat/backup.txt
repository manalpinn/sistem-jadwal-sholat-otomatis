void updateLCD()
{
  // Jangan update jika LCD sedang terkunci (pesan sementara)
  if (lcdLocked)
    return;

  // ===== Ambil waktu sekarang =====
  struct tm tm_info;
  if (rtcAvailable)
  {
    DateTime now = rtc.now();
    tm_info.tm_year = now.year() - 1900;
    tm_info.tm_mon = now.month() - 1;
    tm_info.tm_mday = now.day();
    tm_info.tm_hour = now.hour();
    tm_info.tm_min = now.minute();
    tm_info.tm_sec = now.second();
  }
  else
  {
    time_t t = time(nullptr);
    localtime_r(&t, &tm_info);
  }

  // Format tanggal & jam
  char datebuf[11]; // "DD/MM/YYYY"
  sprintf(datebuf, "%02d/%02d/%04d", tm_info.tm_mday, tm_info.tm_mon + 1, tm_info.tm_year + 1900);

  char timebuf[9]; // "HH:MM:SS"
  sprintf(timebuf, "%02d:%02d:%02d", tm_info.tm_hour, tm_info.tm_min, tm_info.tm_sec);

  // ===== Baris 0: Kota =====
  String kotaLine = "Kota :" + kotaNama;
  if (kotaLine.length() > 20)
    kotaLine = kotaLine.substring(0, 20);
  if (kotaLine != lastKota)
  {
    lcd.setCursor(0, 0);
    lcd.print(kotaLine);
    for (int i = kotaLine.length(); i < 20; i++)
      lcd.print(" ");
    lastKota = kotaLine;
  }

  // ===== Baris 1: Tanggal =====
  String dateLine = "Tgl  :" + String(datebuf);
  if (dateLine.length() > 20)
    dateLine = dateLine.substring(0, 20);
  if (dateLine != lastDate)
  {
    lcd.setCursor(0, 1);
    lcd.print(dateLine);
    for (int i = dateLine.length(); i < 20; i++)
      lcd.print(" ");
    lastDate = dateLine;
  }

  // ===== Baris 2: Jam =====
  String timeLine = "Jam  :" + String(timebuf);
  if (timeLine.length() > 20)
    timeLine = timeLine.substring(0, 20);
  if (timeLine != lastTime)
  {
    lcd.setCursor(0, 2);
    lcd.print(timeLine);
    for (int i = timeLine.length(); i < 20; i++)
      lcd.print(" ");
    lastTime = timeLine;
  }

  // ===== Baris 3: Jadwal sholat =====
  if (isJadwalAvailable())
  {
    JadwalToday jt = getTodayJadwal();
    String listSholat[5] = {jt.subuh, jt.dzuhur, jt.ashar, jt.maghrib, jt.isya};
    String names[5] = {"Sbh", "Dzr", "Ash", "Mgb", "Isy"};

    int nextIdx = 0;
    for (int i = 0; i < 5; i++)
    {
      int h = listSholat[i].substring(0, 2).toInt();
      int m = listSholat[i].substring(3, 5).toInt();
      if ((tm_info.tm_hour < h) || (tm_info.tm_hour == h && tm_info.tm_min < m))
      {
        nextIdx = i;
        break;
      }
      if (i == 4)
        nextIdx = 0;
    }
    int lastIdx = (nextIdx == 0) ? 4 : nextIdx - 1;

    String jadwalLine = names[lastIdx] + ":" + listSholat[lastIdx] + " " +
                        names[nextIdx] + ":" + listSholat[nextIdx];
    if (jadwalLine.length() > 20)
      jadwalLine = jadwalLine.substring(0, 20);

    if (jadwalLine != lastJadwal)
    {
      lcd.setCursor(0, 3);
      lcd.print(jadwalLine);
      for (int i = jadwalLine.length(); i < 20; i++)
        lcd.print(" ");
      lastJadwal = jadwalLine;
    }
  }
}

String urlencode(const String &str)
{
  String encoded = "";
  char c;
  char code0;
  char code1;
  for (int i = 0; i < str.length(); i++)
  {
    c = str.charAt(i);
    if (isalnum(c))
    {
      encoded += c;
    }
    else
    {
      encoded += '%';
      code0 = (c >> 4) & 0xF;
      code1 = c & 0xF;
      encoded += (code0 > 9) ? (code0 - 10 + 'A') : (code0 + '0');
      encoded += (code1 > 9) ? (code1 - 10 + 'A') : (code1 + '0');
    }
  }
  return encoded;
}

void playAdzan(int index)
{
  JadwalToday jt = getTodayJadwal();
  String jam;

  // ===== Pilih jam berdasarkan index =====
  switch (index)
  {
  case 0:
    jam = jt.subuh;
    break;
  case 1:
    jam = jt.dzuhur;
    break;
  case 2:
    jam = jt.ashar;
    break;
  case 3:
    jam = jt.maghrib;
    break;
  case 4:
    jam = jt.isya;
    break;
  default:
    return; // index invalid
  }

  if (jam == "")
    return; // jadwal kosong, tidak ada adzan

  int track = index + 1; // Track DFPlayer

  // ===== Buzzer 3x sesuai config =====
  if (buzzerEnabled)
  {
    for (int i = 0; i < 3; i++)
    {
      digitalWrite(BUZZER_PIN, HIGH);
      delay(150);
      digitalWrite(BUZZER_PIN, LOW);
      delay(150);
    }
  }

  // ===== LOCK LCD =====
  lcdLocked = true;
  lcdLockUntil = millis() + 60000; // lock 60 detik

  // ===== TAMPILKAN LCD 20x4 =====
  lcd.setCursor(0, 0);
  lcd.print("WAKTU SHOLAT        ");

  lcd.setCursor(0, 1);
  lcd.print(namaSholat[index]); // nama sholat sesuai index
  lcd.print("      ");
  lcd.print(jam);
  lcd.print("   ");

  lcd.setCursor(0, 2);
  lcd.print("Adzan Berlangsung  ");

  lcd.setCursor(0, 3);
  lcd.print("Segera Tunaikan    ");

  // ===== PLAY AUDIO DFPlayer =====
  dfPlayer.volume(dfVolume); // gunakan volume terbaru dari config
  dfPlayer.play(track);
}

// ===== SETUP =====
void setup()
{
  Serial.begin(115200);
  delay(100);
  Serial.println("\n=== ESP32 Jadwal Sholat Booting ===");

  // ===== PIN =====
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  pinMode(BUZZER_PIN, OUTPUT);
  digitalWrite(BUZZER_PIN, LOW);

  // ===== LittleFS =====
  Serial.println("Mounting LittleFS...");
  if (!LittleFS.begin(true))
  {
    Serial.println("❌ LittleFS mount failed!");
    lcd.clear();
    lcd.print("FS mount failed!");
    return;
  }
  Serial.println("✅ LittleFS mounted OK");

  // ===== Buat file default jika tidak ada =====
  const char *files[] = {"/index.html", "/style.css", "/script.js", "/kota.json", "/jadwal.json", "/config.json"};
  for (int i = 0; i < sizeof(files) / sizeof(files[0]); i++)
  {
    if (!LittleFS.exists(files[i]))
    {
      Serial.printf("⚠️ %s NOT found, creating default...\n", files[i]);
      File f = LittleFS.open(files[i], "w");
      if (f)
      {
        if (strcmp(files[i], "/kota.json") == 0)
          f.print("{\"id\":0,\"nama\":\"\"}");
        else if (strcmp(files[i], "/jadwal.json") == 0)
          f.print("{\"data\":{\"id\":0,\"lokasi\":\"\",\"daerah\":\"\",\"jadwal\":[]}}");
        else if (strcmp(files[i], "/config.json") == 0)
          f.print("{\"ssid\":\"\",\"password\":\"\",\"volume\":25,\"buzzerEnabled\":true,\"lcdMode\":0}");
        f.close();
        Serial.printf("ℹ️ %s created\n", files[i]);
      }
    }
    else
    {
      Serial.printf("✅ %s found\n", files[i]);
    }
  }

  // ===== Inisialisasi I2C =====
  Wire.begin(LCD_SDA, LCD_SCL);   // Untuk LCD
  I2CRTC.begin(RTC_SDA, RTC_SCL); // Untuk RTC
  I2CRTC.setClock(400000);        // I2C speed

  // ===== Load konfigurasi =====
  loadConfig(); // wifiSSID, wifiPassword, dfVolume, buzzerEnabled, lcdMode

  // ===== RTC =====
  rtcAvailable = rtc.begin(&I2CRTC);

  // ===== LCD =====
  lcd.init();
  lcd.backlight();
  lcd.clear();
  Serial.println("✅ LCD initialized");

  // ===== Setup WiFi =====
  setupWiFi();

  // ===== Ambil waktu dari RTC =====
  DateTime now;
  bool rtcOk = false;

  if (rtcAvailable)
  {
    now = rtc.now();
    if (!rtc.lostPower() && now.year() >= 2023)
    {
      rtcOk = true;
      Serial.printf("Waktu RTC awal: %04d-%02d-%02d %02d:%02d:%02d\n",
                    now.year(), now.month(), now.day(),
                    now.hour(), now.minute(), now.second());
    }
    else
    {
      // RTC tidak valid → set ke waktu compile
      rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
      now = rtc.now();
      rtcOk = true;
      Serial.println("RTC tidak valid → diset ke waktu compile");
      lcd.clear();
      lcd.setCursor(0, 0);
      lcd.print("RTC reset");
      lcd.setCursor(0, 1);
      lcd.print("Waktu compile");
      delay(1000);
      lcd.clear();
    }
  }

  // ===== Sinkronisasi NTP hanya jika perlu =====
  bool timeReady = rtcOk;

  if (WiFi.status() == WL_CONNECTED)
  {
    Serial.println("Cek NTP untuk sinkronisasi...");

    // Ambil waktu NTP
    configTime(GMT_OFFSET_SEC, DAYLIGHT_OFFSET_SEC, NTP_SERVER);
    time_t ntpTime = 0;
    uint32_t start = millis();
    while (millis() - start < 5000)
    { // max 5 detik tunggu NTP
      ntpTime = time(nullptr);
      if (ntpTime > 1600000000)
        break;
      delay(200);
    }

    if (ntpTime > 1600000000)
    {
      DateTime ntpDateTime(ntpTime + GMT_OFFSET_SEC);
      Serial.printf("Waktu NTP: %04d-%02d-%02d %02d:%02d:%02d\n",
                    ntpDateTime.year(), ntpDateTime.month(), ntpDateTime.day(),
                    ntpDateTime.hour(), ntpDateTime.minute(), ntpDateTime.second());

      // Hitung selisih detik antara RTC dan NTP
      long diff = abs((long)ntpDateTime.unixtime() - (long)now.unixtime());
      Serial.printf("Selisih RTC vs NTP: %ld detik\n", diff);

      if (diff > 10)
      { // misal >10 detik → update RTC
        rtc.adjust(ntpDateTime);
        now = rtc.now();
        Serial.println("RTC diperbarui dari NTP");
        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("RTC update NTP");
        lcd.setCursor(0, 1);
        lcd.print(String(now.hour()) + ":" + (now.minute() < 10 ? "0" : "") + String(now.minute()));
        delay(1000);
        lcd.clear();
      }
      else
      {
        Serial.println("RTC dan NTP sama → tidak perlu update");
      }

      timeReady = true;
    }
    else
    {
      Serial.println("❌ NTP gagal");
    }
  }

  // ===== Pesan jika waktu tetap tidak siap =====
  if (!timeReady)
  {
    IPAddress ip = WiFi.status() == WL_CONNECTED ? WiFi.localIP() : WiFi.softAPIP();
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Waktu belum siap");
    lcd.setCursor(0, 1);
    lcd.print("Kunjungi IP:");
    lcd.setCursor(12, 1);
    lcd.print(ip);
    Serial.printf("Silakan kunjungi IP perangkat: %s\n", ip.toString().c_str());
  }

  // ===== Gunakan RTC sebagai waktu sistem =====
  if (rtcAvailable && timeReady)
  {
    now = rtc.now();
    Serial.printf("Waktu sistem final (dari RTC): %04d-%02d-%02d %02d:%02d:%02d\n",
                  now.year(), now.month(), now.day(),
                  now.hour(), now.minute(), now.second());
  }

  // ===== DFPlayer =====
  Serial.println("Initializing DFPlayer...");
  dfSerial.begin(9600, SERIAL_8N1, DFPLAYER_RX, DFPLAYER_TX);
  delay(500);
  if (!dfPlayer.begin(dfSerial))
  {
    Serial.println("❌ DFPlayer not detected!");
  }
  else
  {
    Serial.println("✅ DFPlayer ready");
    dfPlayer.volume(dfVolume);
    Serial.printf("DFPlayer volume set to %d\n", dfVolume);
  }

  // ===== Load kota / jadwal =====
  loadKota();
  if (!loadJadwal())
  {

    // ===== Tampilan gagal load jadwal dengan lcdLocked =====
    Serial.println("⚠️ Failed to load jadwal.json");

    // ===== Lock LCD sementara 3 detik =====
    lcdLocked = true;
    lcdLockUntil = millis() + 3000;

    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Jadwal kosong     "); // baris 0, max 20 char

    lcd.setCursor(0, 1);
    lcd.print("Update via web   "); // baris 1, petunjuk user
    lcd.setCursor(0, 2);
    lcd.print("                    "); // baris 2 kosongkan
    lcd.setCursor(0, 3);
    lcd.print("                    "); // baris 3 kosongkan

    delay(1500); // optional, bisa dihapus kalau loop utama pakai lcdLockUntil
  }
  else
  {
    Serial.println("✅ jadwal.json loaded");
    JadwalToday jt = getTodayJadwal();
    Serial.printf("Today's Subuh:%s Dzuhur:%s Ashar:%s Maghrib:%s Isya:%s\n",
                  jt.subuh.c_str(), jt.dzuhur.c_str(), jt.ashar.c_str(),
                  jt.maghrib.c_str(), jt.isya.c_str());
  }

  // ===== HTTP Server =====
  server.on("/", HTTP_GET, []()
            {
                  if (LittleFS.exists("/index.html"))
                  {
                      File f = LittleFS.open("/index.html", "r");
                      server.streamFile(f, "text/html");
                      f.close();
                  }
                  else
                  {
                      server.send(200, "text/plain", "Web file not found");
                  } });

  server.onNotFound([]()
                    {
                          String path = server.uri();
                          if (LittleFS.exists(path))
                          {
                              File f = LittleFS.open(path, "r");
                              if (path.endsWith(".css"))
                                  server.streamFile(f, "text/css");
                              else if (path.endsWith(".js"))
                                  server.streamFile(f, "application/javascript");
                              else
                                  server.streamFile(f, "text/plain");
                              f.close();
                          } });

  handleAPI(); // Setup semua endpoint API
  server.begin();
  Serial.println("✅ HTTP server started on port 80");

  // ===== INFO IP DEVICE =====
  if (WiFi.status() == WL_CONNECTED)
  {
    Serial.printf("📡 Device IP (STA): %s\n",
                  WiFi.localIP().toString().c_str());
  }
  else
  {
    IPAddress apIP = WiFi.softAPIP();
    Serial.printf("📡 Device IP (AP): %s\n",
                  apIP.toString().c_str());
  }
}
// ===== LOOP =====

void loop()
{
  server.handleClient();
  buttonHandler();

  static int lastStatus = -1; // 0=Belum ada SSID, 1=WiFi gagal, 2=WiFi connect tapi kota belum, 3=Jadwal kosong, 4=Normal
  static unsigned long lastMinuteCheck = 0;
  static unsigned long lastSecondUpdate = 0;

  // ===== Tentukan status =====
  int status = -1;

  if (wifiSSID.length() == 0)
  {
    // Belum ada SSID → tampilkan setup AP
    status = 0;
  }
  else if (WiFi.status() != WL_CONNECTED)
  {
    // WiFi gagal connect, tapi SSID sudah ada → tetap tampil jadwal normal
    status = 4; // normal / jadwal
  }
  else if (!isKotaConfigured())
  {
    status = 2; // WiFi connect & kota belum set
  }
  else if (!isJadwalAvailable())
  {
    status = 3; // Kota ada tapi jadwal kosong
  }
  else
  {
    status = 4; // Normal
  }

  // ===== LCD terkunci sementara =====
  if (lcdLocked && millis() > lcdLockUntil)
  {
    lcdLocked = false;
    lastKota = "";
    lastDate = "";
    lastTime = "";
    lastJadwal = "";
  }

  // ===== Update LCD hanya jika status berubah =====
  if (status != lastStatus && !lcdLocked)
  {
    lcd.clear();

    if (status == 0)
    {
      lcd.setCursor(0, 0);
      lcd.print("Hubungkan WiFi");
      lcd.setCursor(0, 1);
      lcd.print("Buka di browser");
      lcd.setCursor(0, 2);
      lcd.print("IP AP: " + WiFi.softAPIP().toString());
      lcd.setCursor(0, 3);
      lcd.print("Set Kota & WiFi");
      Serial.println("⚠️ Belum ada SSID tersimpan");
    }
    else if (status == 2)
    {
      lcd.setCursor(0, 0);
      lcd.print("WiFi Terhubung     ");
      lcd.setCursor(0, 1);
      lcd.print("Set Kota via IP    ");
      IPAddress ip = (WiFi.getMode() & WIFI_AP) ? WiFi.softAPIP() : WiFi.localIP();
      lcd.setCursor(0, 2);
      lcd.print(ip.toString());
      lcd.setCursor(0, 3);
      lcd.print("Buka browser       ");
      Serial.println("⚠️ Kota belum diset");
    }
    else if (status == 3)
    {
      lcd.setCursor(0, 0);
      lcd.print("Jadwal kosong      ");
      lcd.setCursor(0, 1);
      lcd.print("Update via web     ");
      lcd.setCursor(0, 2);
      lcd.print("                    ");
      lcd.setCursor(0, 3);
      lcd.print("                    ");
      Serial.println("⚠️ Jadwal belum ada");
    }
    else
    {
      // status 4 → normal
      updateLCD();
    }

    lastStatus = status;
  }

  // ===== Update detik setiap 1 detik =====
  if (millis() - lastSecondUpdate >= 1000)
  {
    lastSecondUpdate = millis();
    // Update detik di semua status normal/jadwal tersedia (3 atau 4)
    if (status == 3 || status == 4)
    {
      updateLCD();
    }
  }

  // ===== Cek adzan setiap menit =====
  if (millis() - lastMinuteCheck >= 60000)
  {
    lastMinuteCheck = millis();

    if (isKotaConfigured() && isJadwalAvailable())
    {
      JadwalToday jt = getTodayJadwal();
      if (shouldTrigger(jt.subuh))
        playAdzan(0);
      if (shouldTrigger(jt.dzuhur))
        playAdzan(1);
      if (shouldTrigger(jt.ashar))
        playAdzan(2);
      if (shouldTrigger(jt.maghrib))
        playAdzan(3);
      if (shouldTrigger(jt.isya))
        playAdzan(4);
    }
  }

  delay(50); // loop cepat agar detik update realtime
}

void createDefaultConfig()
{
  Serial.println("Creating default config.json");

  DynamicJsonDocument doc(256);
  doc["volume"] = 25;
  doc["buzzerEnabled"] = true;
  doc["lcdMode"] = 0;
  doc["ssid"] = "";
  doc["password"] = "";

  File f = LittleFS.open(LITTLEFS_CONFIG, "w");
  if (!f)
  {
    Serial.println("Failed to create config.json");
    return;
  }
  serializeJson(doc, f);
  f.close();
}

void loadConfig()
{
  if (!LittleFS.exists(LITTLEFS_CONFIG))
  {
    Serial.println("Config file not found, using defaults");
    return;
  }
  File f = LittleFS.open(LITTLEFS_CONFIG, "r");
  DeserializationError err = deserializeJson(configDoc, f);
  f.close();
  if (err)
  {
    Serial.println("Failed to read config.json");
    return;
  }
  dfVolume = configDoc["volume"] | 25;
  buzzerEnabled = configDoc["buzzerEnabled"] | true;
  lcdMode = configDoc["lcdMode"] | 0;
  wifiSSID = String(configDoc["ssid"].as<const char *>());
  wifiPassword = String(configDoc["password"].as<const char *>());
  Serial.println("Config loaded:");
  Serial.printf(" - volume: %d\n", dfVolume);
  Serial.printf(" - buzzerEnabled: %s\n", buzzerEnabled ? "true" : "false");
  Serial.printf(" - lcdMode: %d\n", lcdMode);
  Serial.printf(" - ssid: %s\n", wifiSSID.c_str());
}

void saveConfig()
{
  File f = LittleFS.open(LITTLEFS_CONFIG, "w");
  if (!f)
  {
    Serial.println("Failed to write config.json");
    return;
  }

  DynamicJsonDocument doc(256);
  doc["volume"] = dfVolume;
  doc["buzzerEnabled"] = buzzerEnabled;
  doc["lcdMode"] = lcdMode;
  doc["ssid"] = wifiSSID;

bool loadJadwal()
{
  if (!LittleFS.exists(LITTLEFS_JADWAL))
  {
    Serial.println("jadwal.json not found, mengunduh jadwal baru...");
    // lanjut ke download di bawah
  }
  else
  {
    File f = LittleFS.open(LITTLEFS_JADWAL, "r");
    if (!f)
    {
      Serial.println("Failed to open jadwal.json, mengunduh jadwal baru...");
    }
    else
    {
      DeserializationError err = deserializeJson(jadwalDoc, f);
      f.close();

      if (err)
      {
        Serial.println("Failed to parse jadwal.json, mengunduh jadwal baru...");
      }
      else
      {
        // =============================
        // Cek bulan terakhir di jadwal
        // =============================
        JsonArray jadwalArray = jadwalDoc["data"]["jadwal"].as<JsonArray>();
        if (jadwalArray.isNull() || jadwalArray.size() == 0)
        {
          Serial.println("Jadwal kosong, mengunduh jadwal baru...");
        }
        else
        {
          const char *lastDateStr = jadwalArray[jadwalArray.size() - 1]["date"];
          int lastYear, lastMonth, lastDay;
          sscanf(lastDateStr, "%d-%d-%d", &lastYear, &lastMonth, &lastDay);

          DateTime now = rtc.now();
          int nowYear = now.year();
          int nowMonth = now.month();

          if (lastYear > 0 && (lastYear < nowYear || (lastYear == nowYear && lastMonth < nowMonth)))
          {
            Serial.println("Bulan sudah lewat, mengunduh jadwal baru...");
            // lanjut ke download
          }
          else
          {
            Serial.println("jadwal.json loaded and valid");
            return true;
          }
        }
      }
    }
  }

  // =============================
  // DOWNLOAD JADWAL LANGSUNG DI SINI
  // =============================
  time_t t = time(nullptr);
  struct tm tm_now;
  localtime_r(&t, &tm_now);
  int tahun = tm_now.tm_year + 1900;
  int bulan = tm_now.tm_mon + 1;

  char url[160];
  snprintf(url, sizeof(url),
           "https://api.myquran.com/v2/sholat/jadwal/%s/%04d/%02d",
           kotaIDStr.c_str(), tahun, bulan);

  Serial.println("Download jadwal → " + String(url));

  WiFiClientSecure client;
  client.setInsecure();
  HTTPClient http;
  http.begin(client, url);
  int httpCode = http.GET();
  if (httpCode != HTTP_CODE_OK)
  {
    Serial.println("Gagal ambil jadwal");
    http.end();
    return false;
  }

  String payload = http.getString();
  http.end();

  if (LittleFS.exists(LITTLEFS_JADWAL))
    LittleFS.remove(LITTLEFS_JADWAL);

  DynamicJsonDocument jd(16 * 1024);
  if (deserializeJson(jd, payload))
  {
    Serial.println("Payload jadwal rusak");
    return false;
  }

  // Simpan ke LittleFS
  File jf = LittleFS.open(LITTLEFS_JADWAL, "w");
  if (!jf)
  {
    Serial.println("Gagal tulis jadwal");
    return false;
  }
  serializeJson(jd, jf);
  jf.close();

  // Load ke RAM
  jadwalDoc = jd;

  Serial.println("Jadwal berhasil diunduh & dimuat");
  return true;
}

// Fungsi trigger adzan
bool shouldTrigger(const String &timeHHMM)
{
  if (timeHHMM.length() != 5)
    return false;

  static bool adzanTriggered[5] = {false, false, false, false, false};
  static int lastDay = -1;

  // === WAKTU DARI RTC ===
  DateTime now = rtc.now();
  int hNow = now.hour();
  int mNow = now.minute();
  int dayNow = now.day();

  // Reset harian
  if (dayNow != lastDay)
  {
    lastDay = dayNow;
    for (int i = 0; i < 5; i++)
      adzanTriggered[i] = false;
  }

  // Ambil jadwal dari JSON
  const char *subuh = jadwalDoc["subuh"];
  const char *dzuhur = jadwalDoc["dzuhur"];
  const char *ashar = jadwalDoc["ashar"];
  const char *maghrib = jadwalDoc["maghrib"];
  const char *isya = jadwalDoc["isya"];

  int idx = -1;
  if (timeHHMM == subuh)
    idx = 0;
  else if (timeHHMM == dzuhur)
    idx = 1;
  else if (timeHHMM == ashar)
    idx = 2;
  else if (timeHHMM == maghrib)
    idx = 3;
  else if (timeHHMM == isya)
    idx = 4;

  if (idx == -1 || adzanTriggered[idx])
    return false;

  // Parse HH:MM
  int h = timeHHMM.substring(0, 2).toInt();
  int m = timeHHMM.substring(3, 5).toInt();

  if (hNow == h && mNow == m)
  {
    adzanTriggered[idx] = true;
    return true;
  }

  return false;
}

void buttonHandler()
{
  bool pressed = digitalRead(BUTTON_PIN) == LOW;
  if (pressed && !buttonDown)
  {
    buttonDown = true;
    buttonPressStart = millis();
  }
  else if (!pressed && buttonDown)
  {
    buttonDown = false;
    unsigned long duration = millis() - buttonPressStart;
    if (duration >= LONG_PRESS_MS)
    {
      Serial.println("Button long-press detected -> start AP Mode");
    }
    else
    {
      lcdMode = (lcdMode + 1) % 3;
      Serial.printf("Button short-press -> lcdMode = %d\n", lcdMode);
    }
  }
}

// ===================== SETUP WIFI =====================
void setupWiFi()
{
  lcd.clear();

  // ====================================================
  // 1️⃣ SET MODE SEKALI SAJA
  // ====================================================
  WiFi.mode(WIFI_AP_STA);
  WiFi.disconnect(true, true);
  delay(200);

  // ====================================================
  // 2️⃣ START AP (SELALU AKTIF)
  // ====================================================
  IPAddress local_ip(192, 168, 4, 1);
  IPAddress gateway(192, 168, 4, 1);
  IPAddress subnet(255, 255, 255, 0);

  WiFi.softAPConfig(local_ip, gateway, subnet);
  WiFi.softAP("Esp32-Sholat", "12345678");

  Serial.printf("✅ AP started: %s\n", WiFi.softAPIP().toString().c_str());

  // ====================================================
  // 1️⃣ TAMPILKAN AP ACTIVE DI BARIS 0 + IP AP DI BARIS 1
  // ====================================================
  // ===== Tampilan AP Aktif =====
  lcdLocked = true;               // kunci LCD sementara
  lcdLockUntil = millis() + 3000; // tampil 3 detik

  lcd.clear();

  for (int i = 0; i < 2; i++)
  {
    digitalWrite(BUZZER_PIN, HIGH);
    delay(150);
    digitalWrite(BUZZER_PIN, LOW);
    delay(150);
  }

  // ===== Baris 0 =====
  lcd.setCursor(0, 0);
  lcd.print("AP Aktif           "); // max 20 char, aman

  // ===== Baris 1 → IP AP =====
  lcd.setCursor(0, 1);
  String ipAP = WiFi.softAPIP().toString();
  String line1 = "IP: " + ipAP;
  if (line1.length() > 20)
    line1 = line1.substring(0, 20); // pastikan tidak overflow
  lcd.print(line1);

  // bersihkan sisa jika kurang dari 20 char
  for (int i = line1.length(); i < 20; i++)
    lcd.print(" ");

  // ===== Baris 2 & 3 kosongkan =====
  lcd.setCursor(0, 2);
  lcd.print("                    "); // 20 spasi
  lcd.setCursor(0, 3);
  lcd.print("                    "); // 20 spasi

  // ====================================================
  // 2️⃣ COBA KONEK KE WIFI JIKA SSID TERSIMPAN
  // ====================================================
  if (wifiSSID.length() > 0)
  {
    Serial.printf("Mencoba koneksi ke WiFi SSID: %s\n", wifiSSID.c_str());

    // Buzzer bunyi 2 kali
    delay(600);

    // Baris 2 → tampilkan SSID (maks 15 karakter)
    lcd.setCursor(0, 2);
    lcd.setCursor(0, 2);  // baris 2
    int maxSSID = 20 - 6; // 6 = panjang "WiFi: "
    String ssidDisplay = wifiSSID.length() > maxSSID ? wifiSSID.substring(0, maxSSID) : wifiSSID;
    String line2 = "WiFi: " + ssidDisplay;
    lcd.print(line2);

    // bersihkan sisa baris
    for (int i = line2.length(); i < 20; i++)
      lcd.print(" ");

    // Baris 3 → status koneksi
    lcd.setCursor(0, 3);
    String line3 = "Menghubungkan...";
    lcd.print(line3);
    for (int i = line3.length(); i < 20; i++)
      lcd.print(" "); // bersihkan sisa

    // Mulai koneksi WiFi
    WiFi.begin(wifiSSID.c_str(), wifiPassword.c_str());
    unsigned long startWiFi = millis();
    bool connected = false;

    while (millis() - startWiFi < 15000)
    {
      if (WiFi.status() == WL_CONNECTED)
      {
        connected = true;
        break;
      }
      delay(500);
      Serial.print(".");
    }
    Serial.println();

    // ====================================================
    // 3️⃣ Hasil koneksi WiFi
    // ====================================================
    lcd.setCursor(0, 3); // bersihkan baris 3
    lcd.print("                    ");

    if (connected)
    {
      Serial.printf("✅ WiFi Tersambung, IP: %s\n", WiFi.localIP().toString().c_str());

      // ===== Bersihkan baris 0–1 dulu (AP lama) =====
      for (int row = 0; row <= 1; row++)
      {
        lcd.setCursor(0, row);
        lcd.print("                    "); // 20 spasi
      }

      // Baris 0 → bisa digunakan untuk judul
      lcd.setCursor(0, 0);
      lcd.print("WiFi Tersambung     "); // pastikan 20 char

      // Baris 1 → tampilkan IP STA
      lcd.setCursor(0, 1);
      String ipSTA = WiFi.localIP().toString();
      String lineIP = "IP: " + ipSTA;
      if (lineIP.length() > 20)
        lineIP = lineIP.substring(0, 20);
      lcd.print(lineIP);
      for (int i = lineIP.length(); i < 20; i++)
        lcd.print(" "); // bersihkan sisa

      // Baris 2 → kosong / instruksi opsional
      lcd.setCursor(0, 2);
      lcd.print("                    ");

      // Baris 3 → status sukses
      lcd.setCursor(0, 3);
      lcd.print("                    "); // bisa juga tulis pesan sukses singkat
    }
    else
    {
      Serial.println("❌ WiFi gagal, AP tetap aktif");
      WiFi.disconnect(true);

      // Bersihkan baris 0–1 juga untuk menghindari tumpukan teks
      for (int row = 0; row <= 1; row++)
      {
        lcd.setCursor(0, row);
        lcd.print("                    ");
      }

      // ===== Tampilkan status gagal sementara =====
      lcdLocked = true;               // kunci LCD sementara
      lcdLockUntil = millis() + 3000; // tampil selama 3 detik

      // Baris 0 → instruksi
      lcd.setCursor(0, 0);
      lcd.print("WiFi gagal     ");
      for (int i = 20; i < 20; i++)
        lcd.print(" "); // bersihkan sisa

      // Baris 1 → status gagal
      lcd.setCursor(0, 1);
      lcd.print("Gunakan AP untuk cfg"); // singkat dan jelas
      for (int i = 16; i < 20; i++)
        lcd.print(" "); // bersihkan sisa

      // Baris 2–3 → bersihkan sisa
      for (int row = 2; row <= 3; row++)
      {
        lcd.setCursor(0, row);
        lcd.print("                    ");
      }
    }
  }
}

void handleAPI()
{
  // Status perangkat
  server.on("/api/status", HTTP_GET, []()
            {
    DateTime now;
    bool waktuTersedia = false;

    // ===== Ambil waktu dari RTC jika tersedia =====
    if (rtcAvailable) {
        now = rtc.now();
        if (!rtc.lostPower() && now.year() >= 2023) {
            waktuTersedia = true;
        } else {
            // RTC tidak valid → set ke waktu compile
            rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
            now = rtc.now();
            waktuTersedia = true;
        }
    }

    // ===== Fallback ke NTP jika RTC tidak valid =====
    if (!waktuTersedia && WiFi.status() == WL_CONNECTED) {
        time_t t = time(nullptr);
        if (t > 1600000000) { // NTP valid
            t += GMT_OFFSET_SEC; // offset WIB
            now = DateTime(t);
            waktuTersedia = true;
        }
    }

    // ===== Format waktu =====
    char waktu[9]; // HH:MM:SS
    if (waktuTersedia) {
        sprintf(waktu, "%02d:%02d:%02d", now.hour(), now.minute(), now.second());
    } else {
        strcpy(waktu, "--:--:--");
    }

    // ===== Buat JSON =====
    String json = "{";
    json += "\"status\":\"ok\",";
    json += "\"wifi\":\"" + WiFi.SSID() + "\",";
    json += "\"kota\":\"" + kotaNama + "\",";
    json += "\"waktu\":\"" + String(waktu) + "\"}";
    
    server.send(200, "application/json", json); });

  // Jadwal hari ini
  server.on("/api/jadwal_today", HTTP_GET, []()
            {

    if (!LittleFS.exists("/jadwal.json")) {
        server.send(404, "application/json", "{\"error\":\"jadwal belum ada\"}");
        return;
    }

    File f = LittleFS.open("/jadwal.json", "r");
    DynamicJsonDocument doc(16 * 1024);
    DeserializationError err = deserializeJson(doc, f);
    f.close();
    if (err) {
        server.send(500, "application/json", "{\"error\":\"json rusak\"}");
        return;
    }

    String today = getTanggalHariIni(); // "YYYY-MM-DD"
    if (today == "1970-01-01") {
        server.send(500, "application/json", "{\"error\":\"waktu belum valid\"}");
        return;
    }

    JsonArray jadwalArr = doc["data"]["jadwal"].as<JsonArray>();
    if (jadwalArr.isNull()) {
        server.send(500, "application/json", "{\"error\":\"struktur jadwal tidak valid\"}");
        return;
    }

    bool found = false;
    for (JsonObject day : jadwalArr) {
        const char* dateStr = day["date"]; // field "date" sesuai JSON
        if (!dateStr) continue;

        if (String(dateStr) == today) {
            DynamicJsonDocument out(512);
            out["imsak"]   = day["imsak"] | "";
            out["subuh"]   = day["subuh"] | "";
            out["dzuhur"]  = day["dzuhur"] | "";
            out["ashar"]   = day["ashar"] | "";
            out["maghrib"] = day["maghrib"] | "";
            out["isya"]    = day["isya"] | "";

            String res;
            serializeJson(out, res);
            server.send(200, "application/json", res);

           Serial.printf("Kirim jadwal hari ini: %s, kota: %s\n", today.c_str(), kotaNama.c_str());

            found = true;
            break;
        }
    }

    if (!found) {
        server.send(404, "application/json", "{\"error\":\"jadwal hari ini tidak ditemukan\"}");
        Serial.printf("No jadwal entry for today: %s\n", today.c_str());
    } });

  // ===== API Search Kota =====
  server.on("/api/search_kota", HTTP_GET, []()
            {
              if (!server.hasArg("kota"))
              {
                server.send(400, "application/json", "{\"error\":\"Parameter 'kota' dibutuhkan\"}");
                return;
              }

              String query = server.arg("kota");
              query.replace(" ", "%20"); // encode spasi

              HTTPClient http;
              String apiURL = "https://api.myquran.com/v2/sholat/kota/cari/" + query;

              http.begin(apiURL); // Initialize HTTP GET
              int httpCode = http.GET();

              if (httpCode > 0)
              {
                if (httpCode == HTTP_CODE_OK)
                {
                  String payload = http.getString();
                  server.send(200, "application/json", payload); // kirim langsung ke browser
                }
                else
                {
                  server.send(httpCode, "application/json", "{\"error\":\"HTTP error\"}");
                }
              }
              else
              {
                server.send(500, "application/json", "{\"error\":\"Failed to connect to API\"}");
              }

              http.end(); // close connection
            });

  // ===== API Set Kota =====
  server.on("/api/set_kota", HTTP_POST, []()
            {
    // 0. VALIDASI BODY
    if (!server.hasArg("plain") || server.arg("plain").isEmpty()) {
        server.send(400, "application/json", R"({"success":false,"error":"Body kosong"})");
        return;
    }

    DynamicJsonDocument doc(512);
    if (deserializeJson(doc, server.arg("plain"))) {
        server.send(400, "application/json", R"({"success":false,"error":"JSON tidak valid"})");
        return;
    }

    // 1. AMBIL DATA KOTA
    String idRaw = doc["id"].as<String>();
    String id = padKotaID(idRaw);
    String nama = doc["nama"].as<String>();

    if (id.isEmpty() || nama.isEmpty()) {
        server.send(400, "application/json", R"({"success":false,"error":"Data kota tidak valid"})");
        return;
    }

    Serial.printf("SET KOTA → %s | %s\n", id.c_str(), nama.c_str());

    // 2. SIMPAN kota.json
    DynamicJsonDocument kotaFile(256);
    kotaFile["id"] = id;
    kotaFile["nama"] = nama;

    File kf = LittleFS.open(LITTLEFS_KOTA, "w");
    if (!kf) {
        server.send(500, "application/json", R"({"success":false,"error":"Gagal menulis kota.json"})");
        return;
    }
    serializeJson(kotaFile, kf);
    kf.close();

    kotaIDStr = id;
    kotaNama = nama;

    // 3. DOWNLOAD JADWAL DARI API
    time_t now;
    if (!getNow(now)) {
        server.send(500, "application/json", R"({"success":false,"error":"Gagal mendapatkan waktu"})");
        return;
    }

    struct tm tm_now;
    localtime_r(&now, &tm_now);
    int tahun = tm_now.tm_year + 1900;
    int bulan = tm_now.tm_mon + 1;

    char url[160];
    snprintf(url, sizeof(url),
             "https://api.myquran.com/v2/sholat/jadwal/%s/%04d/%02d",
             kotaIDStr.c_str(), tahun, bulan);

    Serial.println("Download jadwal → " + String(url));

    WiFiClientSecure client;
    client.setInsecure();

    HTTPClient http;
    http.begin(client, url);
    int httpCode = http.GET();

    if (httpCode != HTTP_CODE_OK) {
        http.end();
        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("Gagal ambil jadwal");
        lcd.setCursor(0, 1);
        lcd.print("Periksa koneksi");
        delay(1000);
        updateLCD();

        server.send(500, "application/json", "{\"success\":false,\"error\":\"Gagal ambil jadwal\"}");
        return;
    }

    String payload = http.getString();
    http.end();

    DynamicJsonDocument jd(16 * 1024);
    if (deserializeJson(jd, payload)) {
        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("Jadwal rusak");
        delay(1000);
        updateLCD();

        server.send(500, "application/json", "{\"success\":false,\"error\":\"Payload jadwal rusak\"}");
        return;
    }

    File jf = LittleFS.open(LITTLEFS_JADWAL, "w");
    if (!jf) {
        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("Gagal tulis jadwal");
        delay(1000);
        updateLCD();

        server.send(500, "application/json", "{\"success\":false,\"error\":\"Gagal menulis jadwal.json\"}");
        return;
    }
    serializeJson(jd, jf);
    jf.close();

    // 4. LOAD JADWAL TERBARU KE RAM
    if (!loadJadwal()) {
        lcdLocked = true;
        lcdLockUntil = millis() + 3000;

        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("Gagal load jadwal");
        lcd.setCursor(0, 1);
        lcd.print("Update via web    ");
        lcd.setCursor(0, 2); lcd.print("                    ");
        lcd.setCursor(0, 3); lcd.print("                    ");

        Serial.println("⚠️ Failed to load jadwal.json");

        server.send(500, "application/json", "{\"success\":false,\"error\":\"Gagal load jadwal\"}");
        return;
    }

    // 5. TAMPILKAN SUKSES DI LCD
    lcdLocked = true;
    lcdLockUntil = millis() + 3000;

    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Kota & jadwal      ");
    lcd.setCursor(0, 1);
    lcd.print("terupdate!          ");
    String namaDisplay = nama.length() > 20 ? nama.substring(0, 20) : nama;
    lcd.setCursor(0, 2);
    lcd.print(namaDisplay);
    for (int i = namaDisplay.length(); i < 20; i++) lcd.print(" ");
    lcd.setCursor(0, 3);
    lcd.print("                    ");

    // 6. RESPONSE JSON SUKSES
    server.send(200, "application/json", "{\"success\":true,\"msg\":\"Kota & jadwal berhasil diperbarui\"}"); });

  //==connect wifi
  server.on("/api/wifi/connect", HTTP_POST, []()
            {
    Serial.println("[API] /api/wifi/connect requested");

    if (!server.hasArg("plain") || server.arg("plain").isEmpty())
    {
        server.send(400, "application/json", "{\"success\":false,\"error\":\"Body kosong\"}");
        return;
    }

    DynamicJsonDocument doc(512);
    if (deserializeJson(doc, server.arg("plain")))
    {
        server.send(400, "application/json", "{\"success\":false,\"error\":\"JSON tidak valid\"}");
        return;
    }

    String ssid = doc["ssid"] | "";
    String password = doc["password"] | "";

    if (ssid.length() == 0)
    {
        server.send(400, "application/json", "{\"success\":false,\"error\":\"SSID kosong\"}");
        return;
    }

    Serial.printf("Connecting to SSID: %s\n", ssid.c_str());

    // ===== Pastikan AP tetap aktif, reset STA =====
    WiFi.disconnect(true, true); // hapus koneksi lama
    delay(100);
    WiFi.mode(WIFI_AP_STA);      // AP+STA aktif
    WiFi.begin(ssid.c_str(), password.c_str());

    unsigned long start = millis();
    bool connected = false;

    while (WiFi.status() != WL_CONNECTED && millis() - start < 15000)
    {
        delay(50);
    }

    // ===== Berhasil connect =====
    if (WiFi.status() == WL_CONNECTED)
    {
        Serial.printf("Connected, IP: %s\n", WiFi.localIP().toString().c_str());

        wifiSSID = ssid;
        wifiPassword = password;
        saveConfig();

        // ===== LCD WiFi Connected =====
        lcdLocked = true;
        lcdLockUntil = millis() + 3000;

        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("WiFi Terhubung     ");

        lcd.setCursor(0, 1);
        String ssidDisplay = ssid.length() > 14 ? ssid.substring(0, 14) : ssid;
        lcd.print("WiFi: " + ssidDisplay);
        for (int i = 6 + ssidDisplay.length(); i < 20; i++) lcd.print(" ");

        lcd.setCursor(0, 2);
        String ipLine = "IP: " + WiFi.localIP().toString();
        lcd.print(ipLine);
        for (int i = ipLine.length(); i < 20; i++) lcd.print(" ");

        lcd.setCursor(0, 3);
        lcd.print("                    ");

        // ===== Response sukses =====
        server.send(200, "application/json",
                    "{\"success\":true,\"ip\":\"" + WiFi.localIP().toString() + "\"}");
    }
    else
    {
        // ===== Gagal connect =====
        Serial.println("Gagal koneksi WiFi");

        // Bersihkan STA agar scan bisa jalan lagi
        WiFi.disconnect(true, true);
        delay(100);
        WiFi.mode(WIFI_AP_STA); // pastikan AP+STA aktif

        // ===== LCD Gagal =====
        lcdLocked = true;
        lcdLockUntil = millis() + 3000;

        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("Gagal koneksi WiFi");

        lcd.setCursor(0, 1);
        lcd.print(ssid.length() > 20 ? ssid.substring(0, 20) : ssid);

        lcd.setCursor(0, 2);
        lcd.print("Periksa password");

        lcd.setCursor(0, 3);
        lcd.print("                    ");

        Serial.println("⚠️ WiFi connect failed");

        server.send(200, "application/json",
                    "{\"success\":false,\"error\":\"Gagal koneksi ke WiFi\"}");
    } });

  // Scan WiFi

  server.on("/api/wifi/scan", HTTP_GET, []()
            {
    Serial.println("[API] /api/wifi/scan requested");

    // Bersihkan hasil scan lama
    WiFi.scanDelete();
    delay(50); // beri jeda sebentar agar scan baru bisa

    // Pastikan AP+STA tetap aktif sebelum scan
    WiFi.mode(WIFI_AP_STA);

    int n = WiFi.scanNetworks(); // blocking scan

    if (n <= 0)
    {
        server.send(200, "application/json", "[]");
        Serial.println("WiFi scan found no networks");
    }
    else
    {
        String json = "[";
        for (int i = 0; i < n; i++)
        {
            json += "{\"ssid\":\"" + WiFi.SSID(i) + "\",\"rssi\":" + String(WiFi.RSSI(i)) + "}";
            if (i < n - 1) json += ",";
        }
        json += "]";
        server.send(200, "application/json", json);
        Serial.printf("WiFi scan complete: %d networks\n", n);
    } });

  // Adzan Manual
  server.on("/api/config", HTTP_POST, []()
            {
    // Ambil body JSON
    String body = server.arg("plain");
    if (body.length() == 0) {
        server.send(400, "application/json", "{\"success\":false,\"error\":\"Body kosong\"}");
        return;
    }

    // Deserialize JSON
    DynamicJsonDocument doc(512); // cukup kecil karena hanya volume + buzzer
    DeserializationError err = deserializeJson(doc, body);
    if (err) {
        server.send(400, "application/json", "{\"success\":false,\"error\":\"JSON invalid\"}");
        return;
    }

    // ===== Update global config =====
    // Gunakan default jika field tidak ada
    dfVolume = doc["volume"] | dfVolume;
    buzzerEnabled = doc["buzzer"] | buzzerEnabled; // ⚡ field sama dengan JS

    // ===== Simpan ke LittleFS =====
    File f = LittleFS.open("/config.json", "w");
    if (!f) {
        server.send(500, "application/json", "{\"success\":false,\"error\":\"Cannot write config\"}");
        return;
    }

    DynamicJsonDocument saveDoc(512);
    saveDoc["volume"] = dfVolume;
    saveDoc["buzzer"] = buzzerEnabled; // ⚡ gunakan "buzzer" agar konsisten dengan JS
    serializeJson(saveDoc, f);
    f.close();

    // Debug serial
    Serial.printf("Config saved: volume=%d, buzzer=%s\n", dfVolume, buzzerEnabled ? "true" : "false");

    // Kirim response sukses
    server.send(200, "application/json", "{\"success\":true}"); });

    String body = "";

    int index = 0; // default Subuh
    if(body.length() > 0){
        DynamicJsonDocument doc(128);
        DeserializationError err = deserializeJson(doc, body);
        if(!err){ // sukses
            index = doc["index"] | 0; // ambil index dari JSON
        } else {
            Serial.println("❌ JSON invalid di /api/adzan/instant");
        }
    }

    // Panggil adzan sesuai index
    playAdzan(index);

    // Response
    server.send(200, "application/json", "{\"status\":\"ok\",\"mode\":\"instant\"}"); });
